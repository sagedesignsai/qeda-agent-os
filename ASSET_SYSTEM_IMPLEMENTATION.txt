TO INTEGRATE (5-MINUTE SETUP)

1. Open: src/renderer/AppLayout.tsx (or your root component)

2. Add import:
   import { AssetReferenceProvider } from '@/contexts/AssetReferenceContext';

3. Wrap your app:
   <AssetReferenceProvider>

   </AssetReferenceProvider>

4. Done! Now any surface can use:
   
   import { useAssetPicker, AssetPicker, AssetReferencesPanel } 
     from '@/components/assets';

     from '@/contexts/AssetReferenceContext';

   const assetPicker = useAssetPicker('my-context-123');
   const { references } = useAssetReferenceContext('my-context-123');

───────────────────────────────────────────────────────────────────────────────
KEY CONCEPTS
───────────────────────────────────────────────────────────────────────────────

CONTEXT ID: Unique string that scopes references
  Examples: 'chat-abc123', 'doc-xyz789', 'task-def456'
  Each surface gets its own ID to prevent collisions

ASSET KINDS: Filter file types when picking
  'media'    → images, video, audio
  'document' → PDFs, Word, text files
  'code'     → source code
  'all'      → everything (default)

PERSISTENCE: Optional auto-save to storage
  localStorage: Simple, default
  IndexedDB: For large datasets
  Custom: Implement your own backend

───────────────────────────────────────────────────────────────────────────────
EXAMPLE USAGE
───────────────────────────────────────────────────────────────────────────────

import { useState } from 'react';
import { 
  useAssetPicker, 
  AssetPicker, 
  AssetReferencesPanel 
} from '@/components/assets';
import { useAssetReferenceContext } 
  from '@/contexts/AssetReferenceContext';

export function MyChatPage({ sessionId, projectId }) {
  const assetPicker = useAssetPicker(sessionId);
  const { references } = useAssetReferenceContext(sessionId);

  const handleSend = () => {
    // Send message with references
    const payload = {
      text: "Hello",
      attachments: references.map(r => ({
        fileName: r.fileName,
        fileKind: r.fileKind,
        path: r.path,
      })),
    };
    // Send to agent...
  };

  return (
    <div>
      <AssetReferencesPanel 
        contextId={sessionId} 
        onAddClick={assetPicker.open} 
      />
      
      <button onClick={assetPicker.open}>Add files</button>
      <button onClick={handleSend}>Send</button>

      <AssetPicker
        open={assetPicker.isOpen}
        onOpenChange={assetPicker.toggle}
        contextId={sessionId}
        kind="all"
        projectId={projectId}
      />
    </div>
  );
}

HOOKS

useAssetReference(contextId, options)
  Returns: { references, addReference, removeReference, ... }
  Use for: Simple per-component tracking

useAssetReferenceContext(contextId)
  Returns: { references, addReference, removeReference, ... }
  Use for: Multi-component surfaces (requires provider)

usePersistentAssets(contextId, storageKey, options)
  Returns: { references, addReference, ... }
  Use for: Sessions that survive page reloads

useAssetPicker(contextId)
  Returns: { isOpen, open, close, toggle }
  Use for: Manage picker dialog state

ASSET REFERENCE TYPE
───────────────────────────────────────────────────────────────────────────────

interface AssetReference {
  id: string;                    // Unique ref ID (aref-...)
  contextId: string;             // Which context owns it
  projectId: string;             // Source project
  fileName: string;              // Display name
  fileKind: string;              // 'image', 'video', etc.
  mediaUrl: string | null;       // Preview/stream URL
  path: string;                  // Absolute filesystem path
  importedAt: number;            // Timestamp
  metadata?: Record<string, any> // Custom data
}

WITH PERSISTENCE
───────────────────────────────────────────────────────────────────────────────

Auto-save to localStorage:

  import { usePersistentAssets } 
    from '@/hooks/use-persistent-assets';

  const { references } = usePersistentAssets(
    'chat-123',
    'chat:123:assets'  // localStorage key
  );

  // Refs auto-save on change, auto-restore on mount

Or with custom storage:

  const { references } = usePersistentAssets(
    contextId,
    storageKey,
    {
      storage: myCustomStorage,  // Your own Storage implementation
      saveDebounceMs: 1000       // Debounce saves (default)
    }
  );
